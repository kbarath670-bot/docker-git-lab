name: Deploy HTML Application

on:
  push:
    branches:
      - main
    paths:
      - "html-app/**"
      - ".github/workflows/deploy-html.yml"

jobs:
  deploy:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Test EC2 SSH connectivity
        run: |
          echo "Testing SSH connectivity to EC2..."
          timeout 10 bash -c "</dev/tcp/${{ secrets.EC2_HOST }}/22" \
            && echo "SUCCESS: EC2 port 22 is reachable" \
            || (echo "FAILED: EC2 port 22 is not reachable" && exit 1)

      - name: Deploy HTML Application to EC2
        uses: appleboy/ssh-action@v1.2.0
        with:
          host: ${{ secrets.EC2_HOST }}
          username: ${{ secrets.EC2_USER }}
          key: ${{ secrets.EC2_SSH_KEY }}
          port: 22

          script: |
            set -e

            cd ~/git-projects/docker-git-lab

            git pull origin main

            cd html-app

            docker stop employee-html || true

            docker rm employee-html || true

            docker rmi employee-html:latest || true

            docker build -t employee-html:latest .

            docker run -d \
              --name employee-html \
              -p 8080:80 \
              employee-html:latest

            docker ps
